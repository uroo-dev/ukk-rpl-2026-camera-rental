Modal Triggers

<button onclick="openModal('modal-id')" class="donezo-btn donezo-btn-primary">Open Modal</button>

<!-- Modal structure -->
<div id="modal-id" class="fixed inset-0 z-50 hidden bg-black/40 backdrop-blur-sm flex items-center justify-center p-4">
  <div class="bg-white rounded-3xl shadow-xl max-w-md w-full p-6 transform transition-all scale-95">
    <div class="flex items-center justify-between mb-4">
      <h3 class="font-bold text-sm">Modal Title</h3>
      <button onclick="closeModal('modal-id')" class="text-gray-400 hover:text-gray-700">
        <i data-lucide="x" class="w-5 h-5"></i>
      </button>
    </div>
    <!-- Content -->
    <div class="flex justify-end gap-2 mt-6">
      <button onclick="closeModal('modal-id')" class="donezo-btn donezo-btn-ghost">Cancel</button>
      <button onclick="closeModal('modal-id')" class="donezo-btn donezo-btn-primary">Confirm</button>
    </div>
  </div>
</div>
