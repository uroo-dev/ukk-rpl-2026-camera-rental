<script>
// Horizontal tab switcher
document.querySelectorAll('[data-h-tab]').forEach(btn => {
  btn.addEventListener('click', () => {
    const target = btn.dataset.hTab;
    btn.parentElement.querySelectorAll('button').forEach(b => {
      b.className = 'pb-3 text-sm font-semibold text-gray-500 border-b-2 border-transparent hover:text-gray-900 hover:border-gray-300 -mb-px transition';
    });
    btn.className = 'pb-3 text-sm font-semibold text-brand-800 border-b-2 border-brand-800 -mb-px';
    btn.closest('.p-6').querySelectorAll('[data-h-pane]').forEach(p => p.classList.add('hidden'));
    const pane = btn.closest('.p-6').querySelector('[data-h-pane="' + target + '"]');
    if (pane) pane.classList.remove('hidden');
  });
});
</script>
