<script>
function updateLiveClock() {
  const now = new Date();
  const h = String(now.getHours()).padStart(2, '0');
  const m = String(now.getMinutes()).padStart(2, '0');
  const s = String(now.getSeconds()).padStart(2, '0');
  const clockEl = document.getElementById('liveClock');
  const dateEl = document.getElementById('liveDate');
  if (clockEl) clockEl.textContent = h + ':' + m + ':' + s;
  if (dateEl) dateEl.textContent = now.toLocaleDateString('en-US', {
    weekday: 'long', year: 'numeric', month: 'long', day: 'numeric'
  });
}
setInterval(updateLiveClock, 1000);
updateLiveClock();
</script>

<script>
function updateCountdown() {
  const container = document.getElementById('countdownTimer');
  if (!container) return;
  const deadline = new Date(container.dataset.deadline).getTime();
  const now = Date.now();
  const diff = Math.max(0, deadline - now);
  const days = Math.floor(diff / 86400000);
  const hours = Math.floor((diff % 86400000) / 3600000);
  const minutes = Math.floor((diff % 3600000) / 60000);
  const seconds = Math.floor((diff % 60000) / 1000);
  container.querySelector('[data-cd="days"]').textContent = String(days).padStart(2, '0');
  container.querySelector('[data-cd="hours"]').textContent = String(hours).padStart(2, '0');
  container.querySelector('[data-cd="minutes"]').textContent = String(minutes).padStart(2, '0');
  container.querySelector('[data-cd="seconds"]').textContent = String(seconds).padStart(2, '0');
}
setInterval(updateCountdown, 1000);
updateCountdown();
</script>
